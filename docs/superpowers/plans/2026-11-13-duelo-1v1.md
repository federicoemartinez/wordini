**Objetivo:** dos amigos, cada uno con su celu, juegan la misma cancha durante 5 minutos. Cada palabra que marca uno queda bloqueada para el otro, y gana el que suma más. Se entra por un link `?vs=CÓDIGO`.

**Arquitectura:**
- **Datos de la partida:** Firebase Realtime Database (plan Spark, gratis y sin tarjeta), usada por REST con `fetch` y escuchada con `EventSource`. Sin SDK y sin `apiKey`: la dirección de la base es pública y `database.rules.json` es lo que protege los datos.
- **Quién marcó primero:** lo decide el servidor con la regla "escribir `words/<palabra>` solo si no existe". También es el servidor el que corta el tiempo, con `now` en las reglas.
- **Misma cancha para los dos:** sale del código de la sala, con el generador determinista que ya existe.
- **`js/duel.js`:** reglas del duelo y red, sin DOM.
- **`js/main.js`:** la pantalla, reutilizando la grilla, el trazado y el teclado de los niveles.

**Stack:**
- Sitio: HTML, CSS y JS (ES2022, módulos nativos). No se agrega ninguna dependencia.
- Tests: `node --test` (Node ≥ 22; probado con v24.13.0).
- Para verificar, solo en el scratchpad y nunca en el repo: `firebase-tools` 15.36.0 (emulador con el proyecto `demo-wordini`, que no pide login; necesita Java, y hay OpenJDK 21) y `playwright-core` con el Chromium headless de `~/.cache/ms-playwright/chromium_headless_shell-1248/`.

```js
// 1-vs-1 duel: rooms in Firebase Realtime Database over REST and EventSource, no SDK. Spec duelo §4.
// The database URL is public on purpose: there is no key, database.rules.json is what protects the data.
import { BIG_FROM, createBoard } from './board.js';
import { score } from './game.js';

export const COUNTDOWN = 3000;
export const LENGTH = 300_000;
const ALPHABET = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789'; // no I, L, O, 0 or 1: easy to read out loud

export const newCode = (rand = Math.random) => Array.from({ length: 5 }, () => ALPHABET[Math.floor(rand() * ALPHABET.length)]).join('');

// Same board for both players: comes from the room code. 5×5 and no gold tiles.
export const duelBoard = (lang, code, words) => ({ ...createBoard(lang, BIG_FROM, words, `duel:${code}`), gold: [] });

export const other = me => (me === 'a' ? 'b' : 'a');
export const wordsOf = (room, who) => Object.keys(room?.words ?? {}).filter(w => room.words[w] === who).sort();

export function scores(room, me) {
  const [mine, theirs] = [wordsOf(room, me), wordsOf(room, other(me))];
  return { mine: score(mine), theirs: score(theirs), nMine: mine.length, nTheirs: theirs.length };
}

// 'waiting' (no rival yet), 'countdown', 'playing' or 'over', at server time `now`.
export function phase(room, now) {
  if (!room?.joined) return 'waiting';
  const start = room.joined + COUNTDOWN;
  return now < start ? 'countdown' : now < start + LENGTH ? 'playing' : 'over';
}

export const timeLeft = (room, now) => Math.max(0, Math.min(LENGTH, room.joined + COUNTDOWN + LENGTH - now));

// Applies one EventSource event to the room and returns it: 'put' replaces what's at `path`, 'patch' replaces
// each child it names. null deletes. A put at "/" replaces the whole room (null = the room doesn't exist).
export function applyEvent(room, type, { path, data }) {
  const at = path.split('/').filter(Boolean);
  const changes = type === 'patch' ? Object.entries(data).map(([k, v]) => [[...at, ...k.split('/')], v]) : [[at, data]];
  for (const [keys, value] of changes) {
    if (!keys.length) {
      room = value;
      continue;
    }
    room ??= {};
    let node = room;
    for (const k of keys.slice(0, -1)) node = node[k] ??= {};
    if (value === null) delete node[keys.at(-1)];
    else node[keys.at(-1)] = value;
  }
  return room;
}
```

```json
          ".write": "!data.exists() && data.parent().child('created').exists()",

        },
        "$other": {
          ".validate": false
```

```js
// A write. Returns the server's answer, or null if the rules rejected it (taken, out of time…). No network throws.
async function send(method, path, body) {
  const res = await fetch(url(path), { method, body: JSON.stringify(body) });
  if (res.status === 401) return null;
  if (!res.ok) throw new Error(`${method} ${path}: ${res.status}`);
  return res.json();
}

export async function getRoom(code) {
  const res = await fetch(url(`rooms/${code}`));
  if (!res.ok) throw new Error(`GET ${code}: ${res.status}`);
  return res.json();
}

// Deletes up to 20 rooms more than a day old. The rules only let you list rooms created before `sweep`, and only
// take a `sweep` more than a day old, so rooms in play never show up (spec §4.5). Best effort: failing is fine.
async function sweep() {
  const before = Date.now() - DAY - 3_600_000; // an hour of margin for a clock running ahead
  if ((await send('PUT', 'sweep', before)) === null) return;
  const res = await fetch(url('rooms', `?orderBy="created"&endAt=${before}&limitToFirst=20`));
  const old = res.ok ? Object.keys((await res.json()) ?? {}) : [];
  if (old.length) await send('PATCH', 'rooms', Object.fromEntries(old.map(code => [code, null])));
}

// Sweeps, then creates a room as 'a'. Returns its code.
export async function create(lang) {
  await sweep().catch(() => {});
  for (let tries = 0; tries < 5; tries++) {
    const code = newCode();
    if (await send('PATCH', `rooms/${code}`, { lang, created: TS })) return code; // null: the code exists, another one
  }
  throw new Error('no room');
}

// Takes the second seat. false if someone already took it.
export const join = async code => (await send('PUT', `rooms/${code}/joined`, TS)) !== null;

// Server time minus local time, from the timestamp that writing t/<me> answers with (spec §4.4).
export async function clockOffset(code, me) {
  const sent = Date.now();
  const server = await send('PUT', `rooms/${code}/t/${me}`, TS);
  return server - (sent + Date.now()) / 2;
}

// Marks `word` as `me`'s. true if it's theirs; false if the rules said no (already marked, or out of time).
export const claim = async (code, word, me) => (await send('PUT', `rooms/${code}/words/${encodeURIComponent(word)}`, me)) !== null;

// Rematch: a new room linked from `next` of the old one, or the rival's if they linked theirs first.
// Returns { code, me }: 'a' = the new room is ours (wait for them), 'b' = it's theirs (join it).
export async function rematch(code, lang, next) {
  if (!next) {
    const fresh = await create(lang);
    if (await send('PUT', `rooms/${code}/next`, fresh)) return { code: fresh, me: 'a' };
    next = await (await fetch(url(`rooms/${code}/next`))).json();
  }
  return { code: next, me: 'b' };
}

// Streams the room: onRoom(room) on every change (null = it doesn't exist). EventSource reconnects by itself and
// Firebase then sends the whole room again. Returns a function that stops it.
export function listen(code, onRoom) {
  let room = null;
  const es = new EventSource(url(`rooms/${code}`));
  for (const type of ['put', 'patch']) es.addEventListener(type, e => onRoom((room = applyEvent(room, type, JSON.parse(e.data)))));
  es.addEventListener('cancel', () => onRoom(null));
  return () => es.close();
}
```

```js
async function enterDuel(code) {
  ++loadSeq; // a level still loading must not take the screen
  if (game) stopClock(); // the level's stopwatch waits for the duel
  board = game = undefined;
  const d = (duel = { code, me: seatIn(code), room: null, offset: 0, theirs: 0, pending: new Set() });
  history.replaceState(null, '', `?vs=${code}`);
  $('#board').replaceChildren();
  $('#hud').hidden = $('#hint-row').hidden = $('#btn-next').hidden = $('#found-box').hidden = $('#btn-duel').hidden = true;
  $('#lang').disabled = true;
  say(t(lang, 'loading'));
  try {
    const room = await getRoom(code);
    if (d !== duel) return;
    if (!room?.created) return duelMessage('duelNoRoom');
    if (!d.me) {
      if (room.joined || !(await join(code))) return duelMessage('duelFull');
      d.me = 'b';
      set(storage, 'wordini:duel', { code, me: 'b' });
    }
    d.offset = await clockOffset(code, d.me);
    const loaded = await fetchWords(room.lang);
    if (d !== duel) return;
    if (!loaded) return say(t(room.lang, 'loadError'));
    lang = room.lang;
    words = loaded;
    document.documentElement.lang = $('#lang').value = lang;
    renderTexts();
    board = duelBoard(lang, code, words);
    renderBoard();
    say('');
    d.room = room;
    d.theirs = scores(room, d.me).theirs;
    d.stop = listen(code, r => onDuelRoom(d, r));
    renderDuel();
  } catch {
    if (d === duel) duelMessage('duelOffline');
  }
}

function onDuelRoom(d, room) {
  if (d !== duel || !board) return;
  if (!room) return d.ended || duelMessage('duelNoRoom');
  d.room = room;
  const { theirs } = scores(room, d.me);
  if (theirs > d.theirs) pop(`+${theirs - d.theirs}`, $('#duel-rival'));
  d.theirs = theirs;
  renderDuel();
}

function renderDuel() {
  const { room, me } = duel;
  const s = scores(room, me);
  $('#duel-hud').hidden = $('#found-box').hidden = false;
  $('#duel-me b').textContent = `${t(lang, 'you')} ${s.mine}`;
  $('#duel-rival b').textContent = `${t(lang, 'rival')} ${s.theirs} (${s.nTheirs})`;
  renderFound(wordsOf(room, me));
  renderDuelClock();
  if (duel.ended) renderDuelEnd();
}

// Every 250 ms: waiting room, 3-2-1, time left and the end. The server's time decides (spec duelo §4.4).
function renderDuelClock() {
  const { room } = duel;
  if (!room || !board) return;
  const now = serverNow();
  const ph = phase(room, now);
  $('#board').classList.toggle('veiled', ph === 'waiting' || ph === 'countdown');
  if (ph === 'waiting') return duelWaiting();
  $('#dlg-duel').close();
  $('#duel-time').textContent = `⏳ ${formatTime(timeLeft(room, now))}`;
  if (ph === 'countdown') say(String(Math.ceil((room.joined + COUNTDOWN - now) / 1000)), 'count');
  else if ($('#word').className === 'count') say('');
  if (ph === 'over') endDuel();
}

function duelWaiting() {
  if ($('#dlg-duel').open) return;
  $('#duel-msg').textContent = t(lang, 'duelWait');
  $('#duel-link').value = duelLink(duel.code);
  $('#duel-invite').hidden = false;
  $('#dlg-duel').showModal();
}

// The duel can't go on (no such room, it's full, no connection): a message and the way back.
function duelMessage(key) {
  board = undefined; // the grid doesn't respond anymore
  $('#board').replaceChildren();
  $('#duel-hud').hidden = $('#found-box').hidden = true;
  say('');
  $('#duel-msg').textContent = t(lang, key);
  $('#duel-invite').hidden = true;
  if (!$('#dlg-duel').open) $('#dlg-duel').showModal();
}

// A word in the duel: checked here first, then the server decides who got it (spec duelo §4.3).
async function duelSubmit(p) {
  const d = duel;
  const word = wordOf(board, p);
  path = [];
  typed = '';
  renderPath();
  if (word.length < 3) return say(p.length > 1 ? t(lang, 'short') : '', 'short');
  const owner = d.room.words?.[word] ?? (d.pending.has(word) ? d.me : null);
  if (!isValid(words, word)) return duelNo(word, 'invalid');
  if (owner) return duelNo(word, owner === d.me ? 'dup' : 'blocked');
  d.pending.add(word);
  say(word, 'live');
  let got;
  try {
    got = await claim(d.code, word, d.me);
  } catch {
    got = null; // no connection
  }
  d.pending.delete(word);
  if (d !== duel) return;
  if (got === null) return duelNo(word, 'duelOffline');
  if (!got) return duelNo(word, d.room.words?.[word] === d.me ? 'dup' : 'blocked');
  (d.room.words ??= {})[word] = d.me; // EventSource confirms it right after; this is just sooner
  say(word, 'good');
  pop(`+${points(word)}`, $('#duel-me'));
  fx.good(word.length);
  renderDuel();
}

function duelNo(word, key) {
  say(`${key === 'blocked' ? '🔒 ' : ''}${word} · ${t(lang, key)}`, key === 'dup' ? 'dup' : 'invalid');
  fx.bad();
}

// Time's up. The result dialog comes back if it gets closed: after the end there's nothing else to do.
function endDuel() {
  if (!$('#dlg-duel-end').open) $('#dlg-duel-end').showModal();
  if (duel.ended) return;
  duel.ended = true;
  tracing = false;
  path = [];
  typed = '';
  renderPath();
  say('');
  const { mine, theirs } = scores(duel.room, duel.me);
  if (mine > theirs) fx.win();
  renderDuelEnd();
}

// Drawn again when a last-second word or the rival's rematch arrives.
function renderDuelEnd() {
  const { room, me } = duel;
  const { mine, theirs } = scores(room, me);
  $('#duel-result').textContent = t(lang, mine > theirs ? 'won' : mine < theirs ? 'lost' : 'tie');
  $('#duel-score').textContent = `${mine} – ${theirs}`;
  $('#duel-mine').textContent = wordsOf(room, me).join(' ');
  $('#duel-theirs').textContent = wordsOf(room, other(me)).join(' ');
  $('#duel-ask').textContent = t(lang, 'rematchAsk');
  $('#duel-ask').hidden = !room.next;
}

async function onRematch() {
  const d = duel;
  try {
    const r = await rematch(d.code, lang, d.room.next);
    if (d !== duel) return;
    if (r.me === 'a') set(storage, 'wordini:duel', r);
    d.stop();
    $('#dlg-duel-end').close();
    enterDuel(r.code);
  } catch {
    $('#duel-ask').textContent = t(lang, 'duelOffline');
    $('#duel-ask').hidden = false;
  }
}

function leaveDuel() {
  duel?.stop?.();
  duel = null;
  $('#dlg-duel').close();
  $('#dlg-duel-end').close();
  $('#duel-hud').hidden = true;
  $('#board').classList.remove('veiled');
  $('#lang').disabled = $('#btn-duel').hidden = false;
  history.replaceState(null, '', location.pathname);
  start(pickLang());
}
```
